 * Load DDEV's connection settings before the project environment policy.
 * The direct include also prevents DDEV appending another include after it.
 */

  require $app_root . '/' . $site_path . '/settings.ddev.php';
}

 * Project-owned environment policy; initialized once by Composer scaffolding.
 */
